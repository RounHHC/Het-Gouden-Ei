      var qs=[].slice.call(t.querySelectorAll('.q')),tas=[].slice.call(t.querySelectorAll('textarea')).filter(function(ta){return !ta.closest('.tblwrap');});

      tas.forEach(function(ta,i){

      });

    var tableRows=[];

    [].slice.call(document.querySelectorAll('.tblwrap tr')).slice(1).forEach(function(tr){

      var tas=tr.querySelectorAll('textarea');

  }

  function dossierToText(d){

    var out='Het Gouden Ei — mijn leerpad\nNaam: '+d.name+'\n\n';

    d.entries.forEach(function(e){out+=e.q+'\n'+e.a+'\n\n';});

    if(d.tableRows.length){

      out+='Samenvattende motieventabel:\n';

      d.tableRows.forEach(function(r){out+='- '+r.label+' — inhoud: '+r.inhoud+' | verband met thema: '+r.verband+'\n';});

      out+='\n';

    }

    out+='Sterren: '+d.stars+' / 5\n';

    d.quizResults.forEach(function(r){out+='Quiz '+r.id+': '+r.r+' / '+r.n+'\n';});

    out+='\nZelfevaluatie:\n';

    d.selfeval.forEach(function(e){out+='- '+e.t+' → '+e.level+'\n';});

    return out;

  }

  function escHtml(s){return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');}

  function dossierToDocHtml(d){

    var h='<html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:w="urn:schemas-microsoft-com:office:word" xmlns="http://www.w3.org/TR/REC-html40">';

    h+='<head><meta charset="utf-8"><title>Het Gouden Ei — dossier</title></head><body style="font-family:Calibri,Arial,sans-serif;font-size:11pt">';

    h+='<h1>Het Gouden Ei — mijn leerpad</h1>';

    h+='<p><b>Naam:</b> '+escHtml(d.name)+'</p>';

    d.entries.forEach(function(e){

      h+='<p><b>'+escHtml(e.q)+'</b><br>'+escHtml(e.a).replace(/\n/g,'<br>')+'</p>';

    });

    if(d.tableRows.length){

      h+='<h2>Samenvattende motieventabel</h2>';

      h+='<table border="1" cellspacing="0" cellpadding="4" style="border-collapse:collapse;width:100%">';

      h+='<tr><th>Motief</th><th>Wat houdt het in?</th><th>Verband met thema</th></tr>';

      d.tableRows.forEach(function(r){h+='<tr><td>'+escHtml(r.label)+'</td><td>'+escHtml(r.inhoud)+'</td><td>'+escHtml(r.verband)+'</td></tr>';});

      h+='</table>';

    }

    h+='<h2>Sterren</h2><p>'+d.stars+' / 5</p>';

    if(d.quizResults.length){

      h+='<h2>Quizresultaten</h2><ul>';

      d.quizResults.forEach(function(r){h+='<li>Quiz '+escHtml(r.id)+': '+r.r+' / '+r.n+'</li>';});

      h+='</ul>';

    }

    h+='<h2>Zelfevaluatie</h2><ul>';

    d.selfeval.forEach(function(e){h+='<li>'+escHtml(e.t)+' \u2192 '+escHtml(e.level)+'</li>';});

    h+='</ul></body></html>';

    return h;

  }
